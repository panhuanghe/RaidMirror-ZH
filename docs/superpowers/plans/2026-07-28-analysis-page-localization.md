function mechanicFixText(row){

  const tips=[];
  if((row.longestMs||0)>=10000)tips.push("更快解除或传递");
  if((row.playersCount||0)>=8)tips.push("收紧团队站位");
  else if((row.playersCount||0)>=4)tips.push("明确点名任务");
  if((row.applications||0)>=10)tips.push("减少重复触发");
  if(!tips.length)tips.push("复盘处理人员和时间点");
  return tips[0]+(tips[1]?` · ${tips[1]}`:"");
}
